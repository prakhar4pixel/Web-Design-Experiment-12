EXPERIMENT 12
Objective: Deploy a full-stack MERN (MongoDB, Express, React, Node.js) application
on a cloud platform like AWS or Heroku.
1. Aim
To build a full-stack MERN notes application and deploy it on a cloud platform (Render).
2. Software Required
• Visual Studio Code (code editor)
• Web Browser (Google Chrome)
• Git and GitHub (version control and online repository)
• Node.js and npm
• Express.js and Mongoose
• React.js 18 (CDN link) and Babel
• MongoDB Atlas (free cloud database)
• Render (cloud platform for deployment)
3. Description
MERN stands for MongoDB, Express.js, React and Node.js. MongoDB stores the data,
Express and Node.js create the REST API, and React shows the user interface. All four
use JavaScript.
In this experiment, the Express server provides the API routes /api/notes for the Create,
Read, Update and Delete operations and saves the notes in MongoDB Atlas using
Mongoose. The React page is placed in the public folder, and the same Express server
serves it, so only one application has to be deployed. The MongoDB link is kept in an
environment variable named MONGO_URI.

4. Source Code
File 1: server.js
require("dotenv").config();
const express = require("express");
const mongoose = require("mongoose");
const app = express();
app.use(express.json());
app.use(express.static("public")); // the React page is in /public
// M = MongoDB
mongoose.connect(process.env.MONGO_URI)
.then(() => console.log("MongoDB connected"))



.catch((err) => console.log("MongoDB error:", err.message));
const Note = mongoose.model("Note", new mongoose.Schema({ text: String }));
//E + N = Express and Node API
app.get("/api/notes", async (req, res) => {
res.json(await Note.find());
});
app.post("/api/notes", async (req, res) => {
res.json(await Note.create({ text: req.body.text }));
});
app.put("/api/notes/:id", async (req, res) => {
res.json(await Note.findByIdAndUpdate(req.params.id, { text: req.body.text }, { new:
true }));
});
app.delete("/api/notes/:id", async (req, res) => {
await Note.findByIdAndDelete(req.params.id);
res.json({ message: "Deleted" });
});
app.listen(process.env.PORT || 3000, () => console.log("Server running on
http://localhost:3000"));

<script type="text/babel">
function App() {
const [notes, setNotes] = React.useState([]);
const [text, setText] = React.useState("");
const [editId, setEditId] = React.useState(null);
// load notes from the server when the page opens
React.useEffect(() => { loadNotes(); }, []);
function loadNotes() {
fetch("/api/notes").then((r) => r.json()).then((data) => setNotes(data));
}
function saveNote() {
if (text.trim() === "") return;
if (editId === null) {
fetch("/api/notes", {
method: "POST",
headers: { "Content-Type": "application/json" },
body: JSON.stringify({ text: text })
}).then(loadNotes);
} else {
fetch("/api/notes/" + editId, {
method: "PUT",
headers: { "Content-Type": "application/json" },
body: JSON.stringify({ text: text })
}).then(loadNotes);
setEditId(null);
}
setText("");
}
function deleteNote(id) {
fetch("/api/notes/" + id, { method: "DELETE" }).then(loadNotes);
}
function editNote(note) {
setText(note.text);
setEditId(note._id);
}
return (
<div className="box">
<h1>MERN Notes</h1>



<input value={text} onChange={(e) => setText(e.target.value)} placeholder="Write a
note" />
<button onClick={saveNote}>{editId === null ? "Add" : "Update"}</button>
<ul>
{notes.map((note) => (
<li key={note._id}>
{note.text}{" "}
<button onClick={() => editNote(note)}>Edit</button>
<button onClick={() => deleteNote(note._id)}>Delete</button>
</li>
))}
</ul>
{notes.length === 0 && <p>No notes yet.</p>}
</div>
);
}
ReactDOM.createRoot(document.getElementById("root")).render(<App />);
</script>
</body>
</html>
